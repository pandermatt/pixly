#define STANDART_TEXT white
#define STANDART_BACKGROUND bgblack
#define STANDART_INACTIVE_TEXT gray
#define STANDART_LOADING lightblue
#define LOADING_LENGHT 30

#define BUFFER_LAENGE 1024


int xPositionForCenteredText(char *text)
{
    return 40-(strlen(text)/2);
}

void printscore(int score, int highscore)
{
    textcolor(STANDART_TEXT | STANDART_BACKGROUND);
    //Score Ausgaben
    gotoxy(2,25);
    printf("%i", score);

    //Highscore ausgeben
    gotoxy(70,25);
    printf("%9i",highscore);
}


int countAsciiInString(char ascii, char *text)
{
    int counter = 0;
    int i = 0;
    for (i = 0; i < strlen(text); i++)
    {
        if (text[i] == ascii)
        {
            counter++;
        }
    }
    return counter;
}

void showMenu(char *titel, char *text, int firstChoice)
{
    int counter = 0;
    char textArray[BUFFER_LAENGE];
    strcpy(textArray, text);
    firstChoice--; //Die Aktuelle auswahl - welches Element ist ausgewählt
    textcolor(STANDART_TEXT | STANDART_BACKGROUND);
    gotoxy(xPositionForCenteredText(titel),5);
    printf("%s", titel); //Den Titel ausgeben

    char *acctualWord;
    acctualWord = strtok(textArray,",");
    while (acctualWord != NULL) //Wort für Wort...
    {
        if (counter == firstChoice)
        {
            textcolor(STANDART_TEXT | STANDART_BACKGROUND);
        }
        else
        {
            textcolor(STANDART_INACTIVE_TEXT | STANDART_BACKGROUND);
        }
        gotoxy(xPositionForCenteredText(acctualWord), 12-(countAsciiInString(',', text)/2)+counter);    //countAsciiInString sort dafür, dass das
        //Menü immer auch in der y-Achse Zentriert ist,
        //egal wie viele Einträge es im Menü hat
        printf("%s",acctualWord); //neuer Eintrag anzeigen
        acctualWord = strtok(NULL, ",");
        counter++;
    }

}

int generateBetterMenu(char *titel, char *text, int firstChoice) //firstChoice ist der Eintrag der Standartmässig ausgewählt ist
{
    //Die einzelnen Einträge (text) werden durch komma getrennt
    unsigned char eingabe;
    showMenu(titel, text, firstChoice); //Menü anzeigen
    while (1)
    {
        if(_kbhit()) //Auf Tastatur eingaben warten
        {
            // Tabehitstencode lesen:
            eingabe = _getch();

            // Prüfen welche Taste gedrückt wurde:
            switch ( eingabe )
            {
            case 72: //rauf
                if (firstChoice > 1)
                {
                    firstChoice--;
                }
                showMenu(titel, text, firstChoice);
                break;
            case 80: //runter
                if (firstChoice < countAsciiInString(',', text)+1)
                {
                    firstChoice++;
                }
                showMenu(titel, text, firstChoice);
                break;
            case 13: //Enter
                return firstChoice;
            }
        }

    }

}


void load(char titel[], double loadingtime, char ascii)
{
    //Variabeln
    int prozent = -1; //
    int i = 0;

    //timer erstellen
    int loadingTimerId;
    loadingTimerId = TimerInit(loadingtime);

    //Bildschirm Clearen
    textcolor(STANDART_TEXT | STANDART_BACKGROUND);
    system("CLS");

    //Titel ausgeben
    gotoxy(xPositionForCenteredText(titel), 12);
    printf("%s", titel);

    //Hintergrund Ladebalken ausgenem
    textcolor(STANDART_LOADING | STANDART_BACKGROUND);
    gotoxy(40-LOADING_LENGHT/2, 14); //Den Ladebalken zentrieren
    for (i = 0; i < LOADING_LENGHT; i++)
    {
        printf("%c", ascii);
    }

    //Ladebalken ausgeben
    while (1)
    {
        if (TimerDown(loadingTimerId))
        {
            //Prozent akktualisieren
            prozent++;
            gotoxy(37,13);
            textcolor(STANDART_TEXT | STANDART_BACKGROUND);
            printf("[%i%%]", prozent);

            //Den neuen Teil des Balkens ausgeben
            gotoxy(40-LOADING_LENGHT/2+1.00*LOADING_LENGHT/100*prozent,14);
            printf("%c", ascii);

            //Funktion beenden
            if (prozent == 100)
            {
                return;
            }
        }
    }
}
