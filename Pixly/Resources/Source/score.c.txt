#include <stdio.h>
#include "consoleio.h"
#include <string.h>
#define TEXT_SIZE 100       //Die Grösse eines Namens
#define BUFFER_LAENGE 1024 //Grösse des Arrays mit den Scores, Die Maximalgrösse einer Zeile
#define STANDART_TEXT white
#define STANDART_BACKGROUND bgblack
#define STANDART_INACTIVE_TEXT gray
#define STANDART_LOADING lightblue //ACHTUNG, Resundant, Muss ich noch anpassen

char dateiname[] = "highscore.txt"; //Hier wird der Score gespeichert
FILE *myFile;
int isLoaded = 0;


struct score
{
    char name[TEXT_SIZE];
    int score;
};

struct score scores[BUFFER_LAENGE]; //In diesem Array sind die Punkte der Spieler gespsichert.
int scoresCount;

int sort_score (void *pp1, void *pp2) //Sortiermethode für qsort
{
    struct score p1 = *((struct score*)pp1);
    struct score p2 = *((struct score*)pp2);

    return p2.score - p1.score;
}

void writeFile() //Schreibt Informatonen aus dem Struct zurück in das File
{
    loadFile();
    myFile = fopen(dateiname,"w");
    qsort(scores, scoresCount, sizeof (struct score), sort_score); //Den Struct sortieren

    int i = 0;
    for (i = 0; i < scoresCount; i++)
    {
        char ausgabe[TEXT_SIZE];
        sprintf(ausgabe, "%s§%i", scores[i].name, scores[i].score);
        lock(ausgabe);
        ausgabe[strlen(ausgabe)+1]=0;
        ausgabe[strlen(ausgabe)]='\n';
        fprintf(myFile, ausgabe);
    }
    fclose(myFile);
}

void lock(char *text)
{
    int i = 0;
    for (i = 0; i < strlen(text); i++)
    {
        text[i] = text[i]+5;
    }
}

void unlock (char *text)
{
    int i = 0;
    for (i = 0; i < strlen(text); i++)
    {
        text[i] = text[i]-5;
    }

}

void loadFile() //File lesen und die Informationen im Struct speichern
{
    if (isLoaded == 0)
    {
        scoresCount = 0;
        char zeile [BUFFER_LAENGE];
        char *tmp;

        myFile = fopen(dateiname,"r");

        if (myFile == NULL) //Existiert das File?
        {
            //File Anlegen
            myFile = fopen(dateiname, "w"); //File anlgen, falls es nicht exisiert
            myFile = fopen(dateiname, "r");
        }


        while (fgets(zeile, BUFFER_LAENGE, myFile) != NULL ) //File zerlegen um im Struct speichern (Komma ist Trennzeichen)
        {
            unlock(zeile);
            tmp = strtok(zeile, "§");
            strcpy(scores[scoresCount].name, tmp);
            tmp = strtok(NULL, "§");
            scores[scoresCount].score = atoi(tmp);
            scoresCount++;
        }

        qsort(scores, scoresCount, sizeof (struct score), sort_score); //Den Struct sortieren
        fclose(myFile);
        isLoaded = 1;
    }
}

int highscore(int AcctualScore) //Gibt Acctualscore zurück falls grösser als highscore
{
    loadFile();
    if (scoresCount > 0);
    {
        if(AcctualScore>scores[0].score)
        {
            return AcctualScore;
        }
        else
        {
            return scores[0].score;
        }
    }
    return 0;
}
